    Basic Tactics
    
          
    
            
    
              
                Assumptions
                intro
                exact
                apply
              
              
    
                Use intro name to move an implication or universal
                binder into the context. Use exact proof when a
                hypothesis or theorem already proves the goal.
                Use revert x to generalize a term variable and its
                dependent hypotheses before induction.
              
    
              
    intro h
exact h
apply theorem_name
    
            
    

            
    
              
                Conjunctions
                split
                .left
                .right
              
              
    
                Use split for conjunction goals. Project existing
                conjunction hypotheses with h.left and
                h.right.
              
    
              
    split
exact h.left
exact h.right
    
            
    

            
    
              
                Disjunctions
                left
                right
                cases
              
              
    
                Use left or right to choose a side of
                an or-goal. Use cases h with to analyze an
                existing disjunction.
              
    
              
    cases h with
| left hp =>
    exact hp
| right hq =>
    exact hq
    
            
    

            
    
              
                Existentials
                exists
                cases
              
              
    
                Use exists witness to provide a witness. Use
                cases h with to unpack a witness and its proof.
              
    
              
    exists x
exact hx

cases h with
| intro x hx =>
    exact hx
    
            
    

            
    
              
                Equality
                refl
                rewrite
              
              
    
                Use refl for goals where both sides compute to the
                same term. Use rewrite h when an equality proof can
                rewrite the current target.
              
    
              
    refl
rewrite h
rewrite -> h
rewrite all h
    
            
    

            
    
              
                Simplification
                simp
                unfold
              
              
    
                Use simp for built-in computation and transparent
                definitions. Use simp at h or simp at *
                for hypotheses.
              
    
              
    simp
simp [lemma]
simp at h
unfold definition_name
    
            
    

            
    
              
                Finite enumeration
                finite
              
              
    
                After importing std/hol/finite@1, use
                finite to prove a concrete list's count,
                membership, or duplicate-freedom. Coverage handles datatypes
                whose constructors take no arguments. Load the three-colour
                example to try it.
              
    
              
    theorem count : F.HasCard(colours, 3) := by
  finite
    
            
    

            
    
              
                Nat and Classical
                induction
                by_cases
                by_contra
              
              
    
                Use induction n with for natural numbers. In
                classical mode, use by_cases h : P or
                by_contra h.
              
    
              
    induction n with
| zero =>
    simp
| succ n ih =>
    simp

by_cases h : P
by_contra h
